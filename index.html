function doGet(e) { var sheetName = (e && e.parameter && e.parameter.sheet) ? e.parameter.sheet : "Program2026"; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); var data = sheet.getDataRange().getValues(); if (data.length < 2) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON); var headers = data[0]; var result = []; for (var i = 1; i < data.length; i++) { var row = data[i]; var item = {}; var isEmpty = true; for (var j = 0; j < headers.length; j++) { item[headers[j]] = row[j] !== undefined ? row[j] : ""; if (row[j] !== "") isEmpty = false; } if (!isEmpty) result.push(item); } return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var contents = ""; if (e && e.postData && e.postData.contents) { contents = e.postData.contents; } else if (e && e.parameter && e.parameter.data) { contents = e.parameter.data; } var postData = (typeof contents === "string") ? JSON.parse(contents) : contents; var sheetName = postData.sheet || "Program2026"; var action = postData.action; // 'create', 'update', 'delete' var payload = postData.data; var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); } var data = sheet.getDataRange().getValues(); if (data.length === 0 || (data.length === 1 && data[0][0] === "")) { if (sheetName === "Program2026") { sheet.appendRow(["รหัสหลักสูตร", "คณะ", "ระดับ", "หลักสูตร", "สาขาวิชา", "ปี", "IQA", "เกณฑ์", "ISCED", "หมายเหตุ"]); } else if (sheetName === "ProgramV2026") { sheet.appendRow(["รหัสหลักสูตร", "คณะ", "ประเภทวิชา", "กลุ่มอาชีพ", "สาขาวิชา", "ปี", "ระดับ", "หมายเหตุ"]); } else if (sheetName === "Plan") { sheet.appendRow(["คณะ", "หลักสูตร", "สาขาวิชา", "ปี", "ระดับ", "วิพากษ์", "ขับเคลื่อนฯ", "สภาวิชาการ", "สภามหาวิทยาลัย", "หมายเหตุ"]); } else if (sheetName === "CISA") { sheet.appendRow(["คณะ", "ระดับ", "หลักสูตร", "สาขาวิชา", "ปี", "CISA status"]); } data = sheet.getDataRange().getValues(); } var headers = data[0]; var keyColIndex = (sheetName === "Program2026" || sheetName === "ProgramV2026") ? headers.indexOf("รหัสหลักสูตร") : headers.indexOf("สาขาวิชา"); if (keyColIndex === -1 && (sheetName === "ProgramV2026" || sheetName === "CISA")) keyColIndex = headers.indexOf("สาขาวิชา"); if (keyColIndex === -1) keyColIndex = 0; if (action === "create") { var newRow = headers.map(function(h) { return payload[h] !== undefined ? payload[h] : ""; }); sheet.appendRow(newRow); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Created" })).setMimeType(ContentService.MimeType.JSON); } else if (action === "update") { var keyValue = payload[headers[keyColIndex]]; var updated = false; for (var i = 1; i < data.length; i++) { if (String(data[i][keyColIndex]).trim() === String(keyValue).trim()) { var updatedRow = headers.map(function(h) { return payload[h] !== undefined ? payload[h] : ""; }); sheet.getRange(i + 1, 1, 1, headers.length).setValues([updatedRow]); updated = true; break; } } if (!updated) { var appendRow = headers.map(function(h) { return payload[h] !== undefined ? payload[h] : ""; }); sheet.appendRow(appendRow); } return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Updated" })).setMimeType(ContentService.MimeType.JSON); } else if (action === "delete") { var delKeyValue = payload[headers[keyColIndex]]; for (var k = 1; k < data.length; k++) { if (String(data[k][keyColIndex]).trim() === String(delKeyValue).trim()) { sheet.deleteRow(k + 1); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Deleted" })).setMimeType(ContentService.MimeType.JSON); } } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Invalid action" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
